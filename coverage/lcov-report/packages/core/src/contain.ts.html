	private async runCommandInContainer(
		command: string,
		args: string[],
		options: { workingDir?: string; env?: Record<string, string>; input?: string } = {},
		containerId: string,
	): Promise<ContainerResult> {
		const execArgs = [
			"exec",
			"-i",
			...(options.workingDir ? ["-w", options.workingDir] : []),
